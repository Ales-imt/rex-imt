DO $$
DECLARE
    r RECORD;
BEGIN
    FOR r IN
        SELECT f.id, f.content, f.user_id, f.season_id,
               s.promotion, NULL::text AS groupe
        FROM feedback f
        LEFT JOIN student s ON s.user_id = f.user_id
        ORDER BY f.created_at DESC
        LIMIT 10
    LOOP
        PERFORM pg_notify('new_feedback', json_build_object(
            'id',        r.id,
            'content',   r.content,
            'user_id',   r.user_id,
            'season_id', r.season_id,
            'promotion', r.promotion,
            'groupe',    r.groupe
        )::text);
        PERFORM pg_sleep(2);
    END LOOP;
END;
$$;
